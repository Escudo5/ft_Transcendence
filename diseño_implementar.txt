function draw42(ctx,cx,cy,w,color,stroke){
  const h=w/1.411,x0=cx-w/2,y0=cy-h/2;
  const F=p=>[x0+p[0]*.5534*w, y0+p[1]*h];
  const T=p=>[x0+(0.6311+p[0]*.3689)*w, y0+p[1]*h];
  ctx[stroke?'strokeStyle':'fillStyle']=color;
  ctx.lineWidth=Math.max(1.5,w*0.006);ctx.lineJoin='miter';
  FOUR.forEach(pts=>{polyPath(ctx,pts,F);ctx[stroke?'stroke':'fill']();});
  TWO.forEach(pts=>{polyPath(ctx,pts,T);ctx[stroke?'stroke':'fill']();});
}

function spaced(ctx,text,cx,y,ls){
  const prev=ctx.textAlign;ctx.textAlign='left';
  const ws=[...text].map(ch=>ctx.measureText(ch).width);
  const total=ws.reduce((a,b)=>a+b,0)+ls*(text.length-1);
  let x=cx-total/2;
  [...text].forEach((ch,i)=>{ctx.fillText(ch,x,y);x+=ws[i]+ls;});
  ctx.textAlign=prev;
}

function holoPatch(ctx,cx,cy,r,rot=0,bright=1){
  ctx.save();ctx.translate(cx,cy);ctx.rotate(rot);
  let g=ctx.createRadialGradient(0,0,r*.2,0,0,r*1.6);
  g.addColorStop(0,`rgba(170,255,255,${.34*bright})`);
  g.addColorStop(.55,`rgba(120,160,255,${.12*bright})`);
  g.addColorStop(1,'rgba(0,0,0,0)');

  for(let i=0;i<n;i++){
    const a=(i/n)*TAU,L=(i%2?.55:1)*r,hue=(i/n*360+185)%360;
    ctx.fillStyle=`hsla(${hue},100%,${bright>0.9?70:60}%,${.95*bright})`;
    ctx.beginPath();
    ctx.moveTo(Math.cos(a-.06)*L,Math.sin(a-.06)*L);
    ctx.lineTo(Math.cos(a+.06)*L,Math.sin(a+.06)*L);
    ctx.lineTo(0,0);ctx.closePath();ctx.fill();

  if(ctx.createConicGradient){

/* ================= GEOMETRÍA INFLADA DEL SOBRE (cuerpo + tira superior) ================= */
const PW=3.3, PH=4.7, TXC=127, TXH=1384;
const VT=1-TXC/TXH;            // línea de rasgado = base del engarce superior
const Y_TEAR=(VT-.5)*PH;

function pillowHalf(v0,v1,NU,NV,yO,sign){
  const TEETH=19,AMP=.062;
  function thick(u,v){
    const uP=Math.pow(Math.max(Math.sin(Math.PI*u),0),.65);
    const edge=sstep(0,.02,v)*sstep(1,.98,v);
    const body=sstep(.055,.22,v)*sstep(.945,.78,v);
    let z=.055*edge*(1-body)*uP + .40*body*uP;                                  // volumen almohada
    z+=edge*(1-body)*uP*.02*Math.pow(Math.sin(u*Math.PI*46),2);                 // estrías del engarce
    z+=body*.011*(Math.sin(u*23+v*31)*.5+Math.sin(u*41-v*17)*.3+Math.sin(u*13+v*57)*.2); // arrugas foil
    return z;
  }
  const pos=[],uvs=[],idx=[];
  for(let i=0;i<=NU;i++)for(let j=0;j<=NV;j++){
    const u=i/NU,v=v0+(v1-v0)*j/NV;
    const x=(u-.5)*PW;
    const tri=Math.abs((u*TEETH)%1-.5)*2;
    let ew=0;
    if(v<.03)ew=1-v/.03;else if(v>.97)ew=(1-v)/.03;
    const y=(v-.5)*PH+AMP*tri*ew*(v<.5?1:-1)-yO;
    pos.push(x,y,sign*thick(u,v));
    uvs.push(u,v);
  }
  const at=(i,j)=>i*(NV+1)+j;
  for(let i=0;i<NU;i++)for(let j=0;j<NV;j++){
    const a=at(i,j),b=at(i,j+1),c=at(i+1,j+1),d=at(i+1,j);
    if(sign>0)idx.push(a,d,c,a,c,b);else idx.push(a,c,d,a,b,c);
  }
  return {pos,uvs,idx};
}
function buildShell(v0,v1,NU,NV,yO=0){
  const f=pillowHalf(v0,v1,NU,NV,yO,1);
  const b=pillowHalf(v0,v1,NU,NV,yO,-1);
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(f.pos.concat(b.pos),3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(f.uvs.concat(b.uvs),2));
  g.setIndex([...f.idx,...b.idx.map(v=>v+f.pos.length/3)]);
  g.computeVertexNormals();
  g.addGroup(0,f.idx.length,0);
  g.addGroup(f.idx.length,b.idx.length,1);
  return g;
}
/* cose normales del cuerpo con la tira en la línea de rasgado (evita costura) */
function stitchSeam(bodyGeo,stripGeo,NU,NVB,NVS){
  const bp=bodyGeo.attributes.normal, sp=stripGeo.attributes.normal;
  const offB=(NU+1)*(NVB+1), offS=(NU+1)*(NVS+1);
  const rows=[]; const stored=[];
  for(let h=0;h<2;h++)for(let i=0;i<=NU;i++){
    const bi=(h?offB:0)+i*(NVB+1)+NVB;
    const si=(h?offS:0)+i*(NVS+1);
    const nx=(bp.getX(bi)+sp.getX(si))/2, ny=(bp.getY(bi)+sp.getY(si))/2, nz=(bp.getZ(bi)+sp.getZ(si))/2;
    const l=Math.hypot(nx,ny,nz)||1;
    bp.setXYZ(bi,nx/l,ny/l,nz/l); sp.setXYZ(si,nx/l,ny/l,nz/l);
    rows.push(si); stored.push(nx/l,ny/l,nz/l);
  }
  bp.needsUpdate=true;sp.needsUpdate=true;
  return {rows,stored};
}

        if(state==='idle'){
          const hit=pick([bodyMesh,stripMesh])[0];
          if(hit){
            const lp=pack.worldToLocal(hit.point.clone());
            if(lp.y>Y_TEAR-.3){
              state='tearing';tear=0;tearTarget=0;
              startX=e.clientX;startY=e.clientY;
              controls.enabled=false;spinTarget=0;
              try{el.setPointerCapture(e.pointerId);}catch(err){}
              cbs.current.onPhase('tearing');
            }else{
              spinVel=4.6;pulseV=1;pulseK=0;
              cbs.current.onPulse();
            }
          }
        }else if(state==='open'){
          const hit=pick(cards)[0];
          if(hit){hit.object.userData.pt=0;pulseV=1;pulseK=0;cbs.current.onPulse();}
        }
      });
      el.addEventListener('pointermove',e=>{
        lastPX=e.clientX;lastPY=e.clientY;
        if(state==='tearing'){
          const dx=e.clientX-startX,dy=e.clientY-startY;
          const f=.5+.5*Math.abs(Math.cos(pack.rotation.y));
          tearTarget=clamp((dx+dy*.35)*.0022*f,0,1);
        }
      });
      el.addEventListener('pointerup',()=>{
        if(state==='tearing'){
          if(tear>.85){completeOpen();}
          else{
            state='idle';tearTarget=0;
            controls.enabled=true;spinTarget=.3;
            cbs.current.onPhase('idle');
          }
        }
      });

      function resize(){
        const a=innerWidth/innerHeight;
        camera.aspect=a;
        camera.fov=a<0.7?54:a>1.05?44:34;
        camera.updateProjectionMatrix();
        renderer.setSize(innerWidth,innerHeight);
        renderer.setPixelRatio(Math.min(devicePixelRatio,2));
      }
      addEventListener('resize',resize);resize();

      /* ---- bucle de animación ---- */
      const clock=new THREE.Clock();
      let perfT=0,first=true;
      renderer.setAnimationLoop(()=>{
        const dt=Math.min(clock.getDelta(),.05);
        perfT+=dt;const t=perfT;
        intro=Math.min(1,intro+dt*.9);
        const ei=1-Math.pow(1-intro,3);

        /* rotación continua (se pausa suavemente mientras se rasga) */
        spinSpeed+=(spinTarget-spinSpeed)*Math.min(1,dt*3.5);
        spinVel*=Math.pow(.14,dt);
        yaw+=spinSpeed*dt+spinVel*dt;
        pack.position.y=.12+Math.sin(t*.9)*.12-(1-ei)*.9;
        pack.rotation.y=yaw+(1-ei)*1.8;
        pack.rotation.z=Math.sin(t*.5)*.03;
        pack.rotation.x=Math.sin(t*.37)*.035;
        pack.scale.setScalar((1+.05*pulseV)*(.92+.08*ei));
        pulseV*=Math.exp(-dt*2.6);pulseK+=dt;

        const em=.85+pulseV*2.6+((state==='tearing')?tear*1.1:0);
        frontMat.emissiveIntensity=backMat.emissiveIntensity=em;
        stripFront.emissiveIntensity=stripBack.emissiveIntensity=em;
        const ir=[140+90*Math.sin(t*.35),520+90*Math.cos(t*.27)];
        frontMat.iridescenceThicknessRange=ir;backMat.iridescenceThicknessRange=ir;
        stripFront.iridescenceThicknessRange=ir;stripBack.iridescenceThicknessRange=ir;

        /* rasgado: muelle hacia el objetivo + auto-completado */
        if(state==='tearing'){
          tear+=(tearTarget-tear)*Math.min(1,dt*14);
          if(tear>.985)completeOpen();
        }else if(state==='idle'&&tear>0){
          tear+=(0-tear)*Math.min(1,dt*6);
          if(tear<.004)tear=0;
        }
        if(state==='tearing'||(state==='idle'&&tear>0)){
          updateStrip(tear,t,tear===0);
          cbs.current.onProgress(clamp(tear,0,1));
        }

        /* vuelo de la tira arrancada */
        if(state==='open'||state==='resetting'){
          if(state==='open'&&fly<1)fly=Math.min(1,fly+dt/1.5);
          if(state==='resetting'){
            fly=Math.max(0,fly-dt/.65);
            openT=Math.max(0,openT-dt*1.5);
            if(fly===0&&openT===0){
              state='idle';tear=0;tearTarget=0;
              updateStrip(0,t,true);
              stripGroup.position.set(0,Y_TEAR,0);
              stripGroup.rotation.set(0,0,0);
              stripFront.opacity=stripBack.opacity=1;
              stripGroup.visible=true;
            }
          }
          const fe=fly*fly*(3-2*fly);
          stripGroup.position.set(0,Y_TEAR+fe*2.1+Math.sin(fe*9)*.05*(1-fe),fe*1.5);
          stripGroup.rotation.set(fe*.55,fe*.5,fe*.3);
          const op=1-sstep(.55,1,fe);
          stripFront.opacity=stripBack.opacity=op;
          stripGroup.visible=fe<1;
        }

        /* cartas: suben escalonadas al abrir, flotan, y responden al click */
        cards.forEach((m,i)=>{
          const p=clamp((openT-DELAY[i])/CDUR,0,1);
          const e=easeOutBack(p);
          const cy=CARD_INSIDE+((Y_TEAR-CARD_H/2+PEEK[i])-CARD_INSIDE)*e;
          m.position.set(CDX[i]+Math.sin(t*.7+i*2.1)*.015*e, cy+Math.sin(t*1.1+i)*.03*e, .06+.045*i);
          m.rotation.set(0,Math.sin(t*.8+i*1.7)*.07*e,(i-1)*.03*e);
          m.scale.setScalar(.97+.03*e);
          const ud=m.userData;
          if(ud.pt>=0){
            ud.pt+=dt*1.5;
            if(ud.pt>1){ud.pt=-1;m.material.emissiveIntensity=0;}
            else m.material.emissiveIntensity=.5*Math.sin(ud.pt*Math.PI);
          }
        });

        /* partículas del estallido */
        if(bT<1.3){
          bT+=dt;const g=3.4;
          for(let i=0;i<BN;i++){
            bVel[i*3+1]-=g*dt;
            bPos[i*3]+=bVel[i*3]*dt;bPos[i*3+1]+=bVel[i*3+1]*dt;bPos[i*3+2]+=bVel[i*3+2]*dt;
          }
          bGeo.attributes.position.needsUpdate=true;
          bMat.opacity=Math.max(0,1-bT/1.3);
          if(bT>=1.3)burst.visible=false;
        }

        /* destello de cremallera */
        const showSpark=(state==='tearing'||((state==='idle'||state==='resetting')&&tear>0))&&tear<.98;
        spark.visible=showSpark;
        if(showSpark){
          spark.position.set((clamp(tear*1.22,0,1)-.5)*PW*.96,Y_TEAR,.2+Math.sin(t*30)*.02);
          const s=.45+.12*Math.sin(t*24)+tear*.15;
          spark.scale.set(s,s,1);
        }

        innerLight.intensity+=(((state==='open')?3+Math.sin(t*2.3)*.5:0)-innerLight.intensity)*Math.min(1,dt*4);

        rings.rotation.z=t*.05;
        ringsMat.opacity=.68+.14*Math.sin(t*1.3)+pulseV*.5;
        if(pulseK<.95){const k=pulseK/.95;pulse.visible=true;
          pulse.scale.setScalar(.5+k*1.9);pulseMat.opacity=(1-k)*.9;}
        else pulse.visible=false;

        const p=dgeo.attributes.position.array;
        for(let i=0;i<ND;i++){
          p[i*3+1]+=dsp[i]*dt;
          if(p[i*3+1]>FLOOR_Y+6.2)p[i*3+1]=FLOOR_Y+.1;
          p[i*3]+=Math.sin(t*.6+i)*0.0012;
        }
        dgeo.attributes.position.needsUpdate=true;

        /* hover sobre la tira (throttle) */
        hoverAcc+=dt;
        if(hoverAcc>.08){
          hoverAcc=0;
          if(state==='idle'&&lastPX!==null){
            setPtr(lastPX,lastPY);
            const hit=pick([bodyMesh,stripMesh])[0];
            const h=!!hit&&pack.worldToLocal(hit.point.clone()).y>Y_TEAR-.3;
            if(h!==hoverStrip){hoverStrip=h;el.classList.toggle('hover-tear',h);cbs.current.onHover(h);}
          }else if(hoverStrip&&state!=='idle'){
            hoverStrip=false;el.classList.remove('hover-tear');cbs.current.onHover(false);
          }
        }

        controls.update();
        renderer.render(scene,camera);
        if(first){first=false;cbs.current.onPhase('idle');}
      });
    })();

    return ()=>{
      disposed=true;
      if(renderer)renderer.setAnimationLoop(null);
      if(controlsRef)controlsRef.dispose();
      if(renderer){
        renderer.dispose();
        const c=mountRef.current&&mountRef.current.querySelector('canvas');
        if(c)c.remove();
      }
      apiRef.current=null;
    };
  },[]);

  return <div id="app" ref={mountRef}/>;
}

/* ================= COMPONENTES DE HUD (React) ================= */
function Loader({done}){
  const [gone,setGone]=useState(false);
  useEffect(()=>{
    if(done){const t=setTimeout(()=>setGone(true),950);return()=>clearTimeout(t);}
  },[done]);
  if(gone)return null;
  return (
    <div id="loader" className={done?'off':''}>
      <img src={logoURL} alt=""/>
      <span>CARGANDO PACK</span>
    </div>
  );
}

function HUD({phase,hover,flash,onReset}){
  const chipTxt=flash?'PULSO HOLO':phase==='open'?'PACK ABIERTO':phase==='tearing'?'ABRIENDO…':'HOLO-ACTIVO';
  return (
    <div id="hud">
      <header>
        <div className="brand">
          <img src={logoURL} alt="42"/>
          <div className="brand-txt"><span>42 MADRID</span><em>NEXT GEN COLLECTION</em></div>
        </div>
        <div className="meta">BOOSTER PACK · ED. 2042<br/><b>HOLO FOIL — Nº 0042</b></div>
      </header>
      <footer>
        <div className={'hint'+(hover?' tearhot':'')}>
          {phase==='open'
            ? <React.Fragment><b>PACK ABIERTO</b> · CLICK EN LAS CARTAS · ARRASTRA PARA ORBITAR</React.Fragment>
            : phase==='tearing'
            ? <React.Fragment><b>SIGUE ARRASTRANDO</b> PARA COMPLETAR LA APERTURA</React.Fragment>
            : <React.Fragment><b>CLICK EN LA TIRA SUPERIOR + ARRASTRA</b> ABRIR&nbsp;&nbsp;·&nbsp;&nbsp;<b>ARRASTRA</b> ROTAR&nbsp;&nbsp;·&nbsp;&nbsp;<b>SCROLL</b> ZOOM</React.Fragment>}
        </div>
        <div className="right">
          {phase==='open'&&
            <button className="reset" onClick={onReset}>
              <svg viewBox="0 0 24 24" fill="none" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round">
                <path d="M3 12a9 9 0 1 0 3-6.7"/><path d="M3 4v5h5"/>
              </svg>
              REINICIAR
            </button>}
          <div className={'chip'+((flash||phase!=='idle')?' hot':'')}>{chipTxt}</div>
        </div>
      </footer>
    </div>
  );
}

/* ================= APP (React) ================= */
function App(){
  const [phase,setPhase]=useState('loading');   // loading | idle | tearing | open
  const [hover,setHover]=useState(false);
  const [flash,setFlash]=useState(false);
  const barRef=useRef(null),pctRef=useRef(null);
  const apiRef=useRef(null),flashT=useRef(0);

  /* el progreso del rasgado (60 fps) se escribe directo en el DOM vía refs: sin re-renders */
  const onProgress=p=>{
    if(barRef.current)barRef.current.style.transform=`scaleX(${p})`;
    if(pctRef.current)pctRef.current.textContent=`${Math.round(p*100)}%`;
  };
  const onPulse=()=>{
    setFlash(true);
    clearTimeout(flashT.current);
    flashT.current=setTimeout(()=>setFlash(false),900);
  };

  return (
    <React.Fragment>
      <Scene3D
        onPhase={setPhase}
        onHover={setHover}
        onPulse={onPulse}
        onProgress={onProgress}
        apiRef={apiRef}/>
      <div className="vignette"/>
      <HUD phase={phase} hover={hover} flash={flash} onReset={()=>apiRef.current&&apiRef.current.reset()}/>
      <div className={'tearbar'+(phase==='tearing'?' on':'')}>
        <span className="lbl">RASGADO</span>
        <div className="track"><div className="fill" ref={barRef}/></div>
        <span className="pct" ref={pctRef}>0%</span>
      </div>
      <Loader done={phase!=='loading'}/>
    </React.Fragment>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
